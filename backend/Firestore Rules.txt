rules_version = '2';

service cloud.firestore {
	match /databases/{database}/documents {
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
  
  match /databases/{database}/documents {
    // Users collection: read by auth'd users, write only by the user themselves
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // Chat collection and subcollections
    match /chat/{chatId} {
      allow read, write: if request.auth != null;
      match /messages/{messageId} {
        allow read, write: if request.auth != null;
      }
      match /typing/{userId} {
        allow read, write: if request.auth != null;
      }
      match /reactions/{reactionId} {
        allow read, write: if request.auth != null;
      }
    }

    // Feedback collection (recursive)
    match /feedback/{document=**} {
      allow read, write: if request.auth != null;
    }

    // Chat metadata
    match /chat_metadata/chat_config {
      allow read, write: if request.auth != null;
    }

    // Squad state and subcollections
    match /squad/state {
      allow read, write: if request.auth != null;
      // Explicit rule for complaints subcollection
      match /complaints/{complaintId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null; // Anyone auth'd can create a complaint
        allow update, delete: if request.auth != null && 
                             request.auth.uid == resource.data.submittedBy; // Only submitter can modify/delete
      }
    }

    // Schedules collection
    match /schedules/{scheduleId} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null;
      allow delete: if request.auth != null && 
                    request.auth.uid == resource.data.player;
    }

    // Invites collection
    match /invites/{inviteId} {
      allow read, write: if request.auth != null;
    }
  }
}

service firebase.storage {
  match /b/{bucket}/o {
    match /chat_audio/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                   request.resource.size < 10 * 1024 * 1024; // 10MB limit
    }
    match /chat_media/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                   request.resource.size < 10 * 1024 * 1024; // 10MB limit
    }
  }
}
